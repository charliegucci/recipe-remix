2. Keep `permissions: pull-requests: write` (needed for PR comment)
3. Replace the deploy step with NuxtHub preview deploy:
   ```yaml
   - name: Deploy preview
     id: deploy
     env:
       NUXT_HUB_PROJECT_KEY: ${{ secrets.NUXT_HUB_PROJECT_KEY }}
     run: |
       OUTPUT=$(npx nuxthub deploy --preview --no-build 2>&1)
       echo "$OUTPUT"
       PREVIEW_URL=$(echo "$OUTPUT" | grep -oE 'https://[^ ]+\.(pages\.dev|nuxt\.dev)' | head -1)
       echo "preview_url=$PREVIEW_URL" >> $GITHUB_OUTPUT
   ```
4. Keep the existing "Comment preview URL on PR" step that uses `actions/github-script@v7`
5. Update the comment body to say "Preview Deployment" and include the URL
